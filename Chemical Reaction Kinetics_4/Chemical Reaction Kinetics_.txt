        for r_name in RealmType:
            realm_mapping[ubp_realm_frequencies.get(r_name.value, 0.0)] = r_name # Simple reverse mapping

        for freq_val, weight_val in ubp_freq_weights.items():
            realm = None
            for r_freq, r_type in realm_mapping.items():
                if abs(r_freq - freq_val) < 1e-6:
                    realm = r_type
                    break
            if realm is None: # Fallback for other constants
                if freq_val == ubp_config_instance.constants.PI: realm = RealmType.ELECTROMAGNETIC
                elif freq_val == ubp_config_instance.constants.PHI: realm = RealmType.COSMOLOGICAL

            self.register_frequency(
                FrequencyWeight(
                    frequency=freq_val,
                    weight=weight_val,
                    source="ubp_constants_module",
                    realm=realm
                )
            )

    def register_frequency(self, freq_weight: FrequencyWeight):
        key = f"{freq_weight.source}_{freq_weight.frequency}"
        self._frequency_registry[key] = freq_weight
        self._cached_f_avg = None
        self._cached_p_gci = None

    def compute_weighted_frequency_average(self) -> float:
        if self.config_gci.cache_enabled and self._cached_f_avg is not None:
            return self._cached_f_avg
        
        if not self._frequency_registry:
            default_freq = get_config().constants.UBP_REALM_FREQUENCIES.get(get_config().default_realm, 1.0)
            self.register_frequency(FrequencyWeight(default_freq, 1.0, "default_fallback"))
            return self.compute_weighted_frequency_average()

        total_weighted_frequency = 0.0
        total_weight = 0.0
        
        for freq_weight in self._frequency_registry.values():
            total_weighted_frequency += freq_weight.weight * freq_weight.frequency
            total_weight += freq_weight.weight
        
        if total_weight == 0:
            return 0.0 # Return 0 if total weight is 0
        
        f_avg = total_weighted_frequency # As per UBP doc: sum, not normalized avg
        
        if self.config_gci.cache_enabled:
            self._cached_f_avg = f_avg
        
        return f_avg

    def compute_global_coherence_index(self, custom_delta_t: Optional[float] = None) -> float:
        delta_t = custom_delta_t if custom_delta_t is not None else self.config_gci.delta_t

            return self._cached_p_gci
        
        f_avg = self.compute_weighted_frequency_average()

        p_gci = math.cos(phase)
        
        if self.config_gci.cache_enabled and custom_delta_t is None:
            self._cached_p_gci = p_gci
        
        return p_gci
